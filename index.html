Koneksi Live Monitoring via Google Spreadsheet (Apps Script)

Agar aktivitas dan kecurangan siswa dari HP Android, iPhone, maupun Laptop langsung muncul di layar Pengawas secara real-time di GitHub Pages, hubungkan Google Spreadsheet Anda dengan langkah singkat berikut:

Langkah 1: Pasang Script di Google Spreadsheet Anda (Cukup 1x)

  1. Buka Google Spreadsheet Anda, lalu klik menu Ekstensi (Extensions) → Apps Script.
  2. Hapus semua kode yang ada, lalu salin dan tempel kode di bawah ini:
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('MONITORING');
  if (!sheet) {
    sheet = ss.insertSheet('MONITORING');
    sheet.appendRow(['sessionId', 'studentClass', 'studentName', 'status', 'activeSubject', 'violations', 'maxViolations', 'remainingSeconds', 'lastViolationReason', 'lastViolationTime', 'forceStop', 'updatedAt']);
  }
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'get';
  var data = sheet.getDataRange().getValues();
  if (action === 'sync') {
    var sid = e.parameter.sessionId;
    var rowIndex = -1;
    var existingForceStop = false;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(sid)) {
        rowIndex = i + 1;
        existingForceStop = (data[i][10] === true || String(data[i][10]).toUpperCase() === 'TRUE');
        break;
      }
    }
    var rowValues = [
      sid,
      e.parameter.studentClass || '',
      e.parameter.studentName || '',
      existingForceStop ? 'STOP PAKSA' : (e.parameter.status || 'AKTIF'),
      e.parameter.activeSubject || '-',
      Number(e.parameter.violations || 0),
      Number(e.parameter.maxViolations || 3),
      e.parameter.remainingSeconds !== '' ? Number(e.parameter.remainingSeconds) : '',
      e.parameter.lastViolationReason || '-',
      e.parameter.lastViolationTime || '-',
      existingForceStop,
      Date.now()
    ];
    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true, forceStop: existingForceStop })).setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'forceStop') {
    var targetSid = e.parameter.sessionId;
    for (var j = 1; j < data.length; j++) {
      if (String(data[j][0]) === String(targetSid)) {
        sheet.getRange(j + 1, 4).setValue('STOP PAKSA');
        sheet.getRange(j + 1, 11).setValue(true);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'delete') {
    var delSid = e.parameter.sessionId;
    for (var k = data.length - 1; k >= 1; k--) {
      if (String(data[k][0]) === String(delSid)) {
        sheet.deleteRow(k + 1);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'clearDone') {
    for (var m = data.length - 1; m >= 1; m--) {
      var st = String(data[m][3]);
      if (st === 'SELESAI' || st === 'WAKTU HABIS') {
        sheet.deleteRow(m + 1);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var list = [];
  for (var r = 1; r < data.length; r++) {
    if (!data[r][0]) continue;
    list.push({
      id: String(data[r][0]),
      studentClass: String(data[r][1]),
      studentName: String(data[r][2]),
      status: String(data[r][3]),
      activeSubject: String(data[r][4]),
      violations: Number(data[r][5] || 0),
      maxViolations: Number(data[r][6] || 3),
      remainingSeconds: data[r][7] !== '' ? Number(data[r][7]) : null,
      lastViolationReason: String(data[r][8] || '-'),
      lastViolationTime: String(data[r][9] || '-'),
      forceStop: (data[r][10] === true || String(data[r][10]).toUpperCase() === 'TRUE'),
      updatedAt: Number(data[r][11] || 0)
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, sessions: list })).setMimeType(ContentService.MimeType.JSON);
}
  1. Klik Terapkan (Deploy) → Deployment baru (New deployment) → pilih jenis Aplikasi web (Web app).
  2. Pada bagian Yang memiliki akses (Who has access), pilih Siapa saja (Anyone), lalu klik Terapkan dan salin URL Aplikasi Web (berakhiran /exec).
  3. Tempelkan URL tersebut ke variabel APPS_SCRIPT_WEB_APP_URL di dalam file HTML di repository GitHub Anda.

Tips: Agar semua HP siswa di GitHub Pages otomatis terhubung tanpa pengaturan ulang, tempelkan URL /exec Anda pada baris let APPS_SCRIPT_WEB_APP_URL = '...' di dalam kode HTML sebelum commit ke GitHub.
