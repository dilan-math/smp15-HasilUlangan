Koneksi Live Monitoring via Google Spreadsheet (Apps Script)

Agar aktivitas dan kecurangan siswa dari HP Android, iPhone, maupun Laptop langsung muncul di layar Pengawas secara real-time di GitHub Pages, hubungkan Google Spreadsheet Anda dengan langkah singkat berikut:

Langkah 1: Pasang Script di Google Spreadsheet Anda (Cukup 1x)

  1. Buka Google Spreadsheet Anda, lalu klik menu Ekstensi (Extensions) → Apps Script.
  2. Hapus semua kode yang ada, lalu salin dan tempel kode di bawah ini:
function doGet(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'Server sibuk, coba lagi' })).setMimeType(ContentService.MimeType.JSON);
  }
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('MONITORING');
    if (!sheet) {
      sheet = ss.insertSheet('MONITORING');
      sheet.appendRow(['sessionId', 'studentClass', 'studentName', 'status', 'activeSubject', 'violations', 'maxViolations', 'remainingSeconds', 'lastViolationReason', 'lastViolationTime', 'forceStop', 'updatedAt', 'awaySeconds']);
    }
    var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'get';
    var data = sheet.getDataRange().getValues();
    if (action === 'sync') {
      var sid = e.parameter.sessionId;
      var rowIndex = -1;
      var existingForceStop = false;
      var existingStatus = '';
      var existingViolations = 0;
      var existingAwaySeconds = 0;
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(sid)) {
          rowIndex = i + 1;
          existingStatus = String(data[i][3] || '');
          existingViolations = Number(data[i][5] || 0);
          existingForceStop = (data[i][10] === true || String(data[i][10]).toUpperCase() === 'TRUE');
          existingAwaySeconds = Number(data[i][12] || 0);
          break;
        }
      }
      var maxV = Number(e.parameter.maxViolations || 10);
      var incomingV = Number(e.parameter.violations || 0);
      var incomingAway = Number(e.parameter.awaySeconds || 0);
      var isAlreadyLocked = existingForceStop || existingStatus.indexOf('TERKUNCI') !== -1 || existingViolations >= maxV;
      var finalViolations = Math.max(existingViolations, incomingV);
      var finalAwaySeconds = Math.max(existingAwaySeconds, incomingAway);
      var finalStatus = e.parameter.status || 'AKTIF';
      var finalForceStop = existingForceStop;
      if (isAlreadyLocked) {
        finalForceStop = true;
        finalStatus = existingStatus.indexOf('STOP PAKSA') !== -1 ? 'STOP PAKSA' : 'TERKUNCI (10x PELANGGARAN)';
      } else if (finalViolations >= maxV) {
        finalForceStop = true;
        finalStatus = 'TERKUNCI (10x PELANGGARAN)';
      }
      var rowValues = [
        sid,
        e.parameter.studentClass || '',
        e.parameter.studentName || '',
        finalStatus,
        e.parameter.activeSubject || '-',
        finalViolations,
        maxV,
        e.parameter.remainingSeconds !== '' ? Number(e.parameter.remainingSeconds) : '',
        e.parameter.lastViolationReason || '-',
        e.parameter.lastViolationTime || '-',
        finalForceStop,
        Date.now(),
        finalAwaySeconds
      ];
      if (rowIndex > 0) {
        sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, forceStop: finalForceStop, locked: isAlreadyLocked || finalViolations >= maxV, status: finalStatus, violations: finalViolations, awaySeconds: finalAwaySeconds })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'checkLock') {
      var chkSid = String(e.parameter.sessionId || '').trim();
      var chkClass = String(e.parameter.studentClass || '').trim().toLowerCase();
      var chkName = String(e.parameter.studentName || '').trim().toLowerCase();
      for (var c = data.length - 1; c >= 1; c--) {
        var rSid = String(data[c][0] || '').trim();
        var rClass = String(data[c][1] || '').trim().toLowerCase();
        var rName = String(data[c][2] || '').trim().toLowerCase();
        if (rSid === chkSid || (chkClass && chkName && rClass === chkClass && rName === chkName)) {
          var cStatus = String(data[c][3] || '');
          var cViolations = Number(data[c][5] || 0);
          var cMax = Number(data[c][6] || 10);
          var cForce = (data[c][10] === true || String(data[c][10]).toUpperCase() === 'TRUE');
          var locked = cForce || cStatus.indexOf('TERKUNCI') !== -1 || cViolations >= cMax;
          if (locked) {
            return ContentService.createTextOutput(JSON.stringify({ ok: true, locked: true, status: cStatus, violations: cViolations })).setMimeType(ContentService.MimeType.JSON);
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, locked: false, status: 'AMAN', violations: 0 })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'unlock') {
      var unlSid = String(e.parameter.sessionId || '').trim();
      var unlClass = String(e.parameter.studentClass || '').trim().toLowerCase();
      var unlName = String(e.parameter.studentName || '').trim().toLowerCase();
      var unlockedCount = 0;
      for (var u = 1; u < data.length; u++) {
        var rowSid = String(data[u][0] || '').trim();
        var rowClass = String(data[u][1] || '').trim().toLowerCase();
        var rowName = String(data[u][2] || '').trim().toLowerCase();
        if (rowSid === unlSid || (unlClass && unlName && rowClass === unlClass && rowName === unlName)) {
          sheet.getRange(u + 1, 4).setValue('DIBUKA ADMIN');
          sheet.getRange(u + 1, 6).setValue(0);
          sheet.getRange(u + 1, 7).setValue(10);
          sheet.getRange(u + 1, 9).setValue('Reset & Buka Kunci oleh Admin');
          sheet.getRange(u + 1, 11).setValue(false);
          sheet.getRange(u + 1, 12).setValue(Date.now());
          unlockedCount++;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, unlockedCount: unlockedCount })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'forceStop') {
      var targetSid = e.parameter.sessionId;
      for (var j = 1; j < data.length; j++) {
        if (String(data[j][0]) === String(targetSid)) {
          sheet.getRange(j + 1, 4).setValue('STOP PAKSA');
          sheet.getRange(j + 1, 11).setValue(true);
          sheet.getRange(j + 1, 12).setValue(Date.now());
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'delete') {
      var delSid = e.parameter.sessionId;
      for (var k = data.length - 1; k >= 1; k--) {
        if (String(data[k][0]) === String(delSid)) {
          sheet.deleteRow(k + 1);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'clearDone') {
      for (var m = data.length - 1; m >= 1; m--) {
        var st = String(data[m][3]);
        if (st === 'SELESAI' || st === 'WAKTU HABIS') {
          sheet.deleteRow(m + 1);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
    }
    var list = [];
    for (var r = 1; r < data.length; r++) {
      if (!data[r][0]) continue;
      list.push({
        id: String(data[r][0]),
        studentClass: String(data[r][1]),
        studentName: String(data[r][2]),
        status: String(data[r][3]),
        activeSubject: String(data[r][4]),
        violations: Number(data[r][5] || 0),
        maxViolations: Number(data[r][6] || 10),
        remainingSeconds: data[r][7] !== '' ? Number(data[r][7]) : null,
        lastViolationReason: String(data[r][8] || '-'),
        lastViolationTime: String(data[r][9] || '-'),
        forceStop: (data[r][10] === true || String(data[r][10]).toUpperCase() === 'TRUE'),
        updatedAt: Number(data[r][11] || 0),
        awaySeconds: Number(data[r][12] || 0)
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true, sessions: list })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
  1. Klik Terapkan (Deploy) → Deployment baru (New deployment) → pilih jenis Aplikasi web (Web app).
  2. Pada bagian Yang memiliki akses (Who has access), pilih Siapa saja (Anyone), lalu klik Terapkan dan salin URL Aplikasi Web (berakhiran /exec).
  3. Tempelkan URL tersebut ke variabel APPS_SCRIPT_WEB_APP_URL di dalam file HTML di repository GitHub Anda.

Tips: Agar semua HP siswa di GitHub Pages otomatis terhubung tanpa pengaturan ulang, tempelkan URL /exec Anda pada baris let APPS_SCRIPT_WEB_APP_URL = '...' di dalam kode HTML sebelum commit ke GitHub.
